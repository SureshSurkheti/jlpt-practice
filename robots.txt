# exam.html, kanji.html, stats.html and practice.html are each one
# shell serving every ?id= / ?k= / level, so each carries
# <meta name="robots" content="noindex, follow">. They are
# deliberately not disallowed here: a crawler blocked by robots.txt
# never fetches the page, so it never sees the noindex, and the URL
# can still surface in results with no snippet. Allowing the fetch
# is what makes the noindex work.
Disallow: /_src/

# The exam papers themselves. The questions are the copyright of
# JEES and the Japan Foundation; this site holds them so a learner
# can sit a paper, not so they can be re-indexed and re-published.
# Search engines index the library page instead.
Disallow: /data/exams/
Disallow: /data/glossary/
Disallow: /data/exams-manual/

# Dictionary build inputs - 57MB of JMdict and KANJIDIC that the

# linked to them either, but nothing stopped a crawler walking in.
